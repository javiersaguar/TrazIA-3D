import json
import os
import subprocess
import time
from pathlib import Path
from datetime import datetime, timezone
ROOT = Path(__file__).resolve().parents[2]
os.chdir(ROOT)
os.environ["PYTHONUTF8"] = "1"
os.environ["PYTHONIOENCODING"] = "utf-8"
os.environ["TRAZIA_ROOT"] = str(ROOT)
os.environ["TRAZIA_RAG_LIBRARY"] = str(ROOT / "generated/rag/biblioteca.json")
# Cada etapa corre sola; pytest se ejecuta tres veces consecutivas sin otras mediciones en paralelo.
out = ROOT / "generated/bench"
records_file = out / "ejecuciones.json"
records = json.loads(records_file.read_text(encoding="utf-8"))
commands = [
    ("bench", ["uv", "run", "python", "-m", "bench", "--detalle"]),
    ("estudio_b", ["uv", "run", "python", "generated/bench/estudio_v1.py", "b"]),
    ("biblioteca", ["uv", "run", "python", "generated/bench/estudio_v1.py", "biblioteca"]),
    ("estudio_variantes", ["uv", "run", "python", "generated/bench/estudio_v1.py", "variantes"]),
    *[(f"pytest_{i}", ["uv", "run", "pytest", "-q"]) for i in range(1,4)],
]
for name, cmd in commands:
    print(f"INICIO {name} {datetime.now(timezone.utc).isoformat()}", flush=True)
    entry = {"stage": name, "command": cmd, "started_at": datetime.now(timezone.utc).isoformat(), "status": "running"}
    records.append(entry)
    records_file.write_text(json.dumps(records, ensure_ascii=False, indent=2), encoding="utf-8")
    started = time.perf_counter()
    with (out / f"{name}.log").open("w", encoding="utf-8") as log:
        result = subprocess.run(cmd, stdout=log, stderr=subprocess.STDOUT, env=os.environ.copy())
    entry.update(seconds=time.perf_counter()-started, returncode=result.returncode, status="finished", finished_at=datetime.now(timezone.utc).isoformat())
    records_file.write_text(json.dumps(records, ensure_ascii=False, indent=2), encoding="utf-8")
    print(f"FIN {name}: exit={result.returncode}, {entry['seconds']:.2f} s", flush=True)
    print((out / f"{name}.log").read_text(encoding="utf-8",errors="replace")[-1800:], flush=True)
print("PRUEBA COMPLETA", flush=True)
